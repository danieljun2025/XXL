const SAMPLE = { TF: 3.0, GM: 3.0, AU: 4.0, SE: 2.0, EC: 6.8, SV: 5.8, CH: 5.8, LS: 2.0 };
function radarSvg(size, labelsOn = true) {
  const C = size / 2, R = size * 0.32, n = 8, MAX = 8.4;
  const pt = (i, r) => { const t = -Math.PI / 2 + (2 * Math.PI * i) / n; return [C + r * Math.cos(t), C + r * Math.sin(t)]; };
  const rings = [1 / 3, 2 / 3, 1].map((f) => `<polygon fill="none" stroke="#d9e2f1" stroke-width="2" points="${ORDER.map((_, i) => pt(i, R * f).join(',')).join(' ')}"/>`).join('');
  const axes = ORDER.map((_, i) => `<line x1="${C}" y1="${C}" x2="${pt(i, R)[0]}" y2="${pt(i, R)[1]}" stroke="#d9e2f1" stroke-width="2"/>`).join('');
  const poly = ORDER.map((c, i) => pt(i, (R * SAMPLE[c]) / MAX).join(',')).join(' ');
  const top = [...ORDER].sort((a, b) => SAMPLE[b] - SAMPLE[a]).slice(0, 3);
  const dots = ORDER.map((c, i) => { const [x, y] = pt(i, (R * SAMPLE[c]) / MAX); return `<circle cx="${x}" cy="${y}" r="${top.includes(c) ? 13 : 7}" fill="#2f6fed" stroke="#fff" stroke-width="4"/>`; }).join('');
  const labels = ORDER.map((c, i) => { const [x, y] = pt(i, R + 62); return `<text x="${x}" y="${y - 8}" text-anchor="middle" font-size="32" font-weight="${top.includes(c) ? 800 : 500}" fill="${top.includes(c) ? '#16202c' : '#6b7788'}">${esc(A[c].name.replace('型', ''))}</text><text x="${x}" y="${y + 30}" text-anchor="middle" font-size="28" fill="${top.includes(c) ? '#2f6fed' : '#8a95a5'}" font-weight="700">${SAMPLE[c].toFixed(1)}</text>`; }).join('');
  return `<svg width="${size}" height="${size}" viewBox="0 0 ${size} ${size}">${rings}${axes}<polygon points="${poly}" fill="rgba(47,111,237,.22)" stroke="#2f6fed" stroke-width="5" stroke-linejoin="round"/>${dots}${labels}</svg>`;
}
